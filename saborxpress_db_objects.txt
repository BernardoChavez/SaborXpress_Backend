-- Función 1: Obtener ventas cobradas en Efectivo para una caja determinada
CREATE FUNCTION dbo.fn_ObtenerVentasEfectivo (@id_caja BIGINT)
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @total DECIMAL(10,2);
    SELECT @total = ISNULL(SUM(monto_total), 0)
    FROM dbo.ventas
    WHERE id_caja = @id_caja AND estado = 'Pagado' AND metodo_pago = 'Efectivo';
    RETURN @total;
END;
GO

-- Función 2: Obtener ventas cobradas por QR para una caja determinada
CREATE FUNCTION dbo.fn_ObtenerVentasQR (@id_caja BIGINT)
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @total DECIMAL(10,2);
    SELECT @total = ISNULL(SUM(monto_total), 0)
    FROM dbo.ventas
    WHERE id_caja = @id_caja AND estado = 'Pagado' AND metodo_pago = 'QR';
    RETURN @total;
END;
GO

-- Función 3: Calcular la diferencia de Efectivo en el cierre de caja
CREATE FUNCTION dbo.fn_CalcularDiferenciaEfectivo (@id_caja BIGINT, @monto_real DECIMAL(10,2))
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @monto_apertura DECIMAL(10,2);
    DECLARE @ventas_efectivo DECIMAL(10,2);
    
    SELECT @monto_apertura = monto_apertura FROM dbo.cajas WHERE id = @id_caja;
    SET @ventas_efectivo = dbo.fn_ObtenerVentasEfectivo(@id_caja);
    
    RETURN @monto_real - (@monto_apertura + @ventas_efectivo);
END;
GO

-- Función 4: Calcular la diferencia de montos QR en el cierre de caja
CREATE FUNCTION dbo.fn_CalcularDiferenciaQR (@id_caja BIGINT, @monto_real_qr DECIMAL(10,2))
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @monto_apertura_qr DECIMAL(10,2);
    DECLARE @ventas_qr DECIMAL(10,2);
    
    SELECT @monto_apertura_qr = ISNULL(monto_apertura_qr, 0) FROM dbo.cajas WHERE id = @id_caja;
    SET @ventas_qr = dbo.fn_ObtenerVentasQR(@id_caja);
    
    RETURN @monto_real_qr - (@monto_apertura_qr + @ventas_qr);
END;
GO

-- Trigger 1: Descontar stock procesado de forma automática al registrar detalles de venta (CU32)
CREATE TRIGGER dbo.trg_DescontarStockAlVender
ON dbo.venta_detalles
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    -- Validar si alguno de los insumos no tiene stock suficiente
    IF EXISTS (
        SELECT 1
        FROM dbo.inventario_procesado ip
        INNER JOIN dbo.recetas r ON ip.id = r.id_procesado
        INNER JOIN inserted i ON r.id_producto = i.id_producto
        WHERE ip.stock < (r.cantidad * i.cantidad)
    )
    BEGIN
        -- Generar mensaje de error detallado
        DECLARE @NombreInsumo NVARCHAR(100);
        SELECT TOP 1 @NombreInsumo = ip.nombre
        FROM dbo.inventario_procesado ip
        INNER JOIN dbo.recetas r ON ip.id = r.id_procesado
        INNER JOIN inserted i ON r.id_producto = i.id_producto
        WHERE ip.stock < (r.cantidad * i.cantidad);

        DECLARE @Msg NVARCHAR(200) = N'ERROR: Stock insuficiente del insumo procesado "' + @NombreInsumo + N'" para completar la venta.';
        RAISERROR(@Msg, 16, 1);
        ROLLBACK TRANSACTION;
        RETURN;
    END;

    -- Descontar el stock si todo es correcto
    UPDATE ip
    SET ip.stock = ip.stock - (r.cantidad * i.cantidad),
        ip.updated_at = GETDATE()
    FROM dbo.inventario_procesado ip
    INNER JOIN dbo.recetas r ON ip.id = r.id_procesado
    INNER JOIN inserted i ON r.id_producto = i.id_producto;
END;
GO

-- Trigger 2: Crear Comanda automática para Cocina al registrar una Venta (CU20)
CREATE TRIGGER dbo.trg_CrearComandaAlVender
ON dbo.ventas
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO dbo.comandas (id_venta, estado, area, created_at, updated_at)
    SELECT id, N'Pendiente', N'Cocina', GETDATE(), GETDATE()
    FROM inserted;
END;
GO

-- Trigger 3: Auditoría automática en bitácora para inventario bruto
CREATE TRIGGER dbo.trg_AuditarInventarioBruto
ON dbo.inventario_bruto
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @accion NVARCHAR(100);
    DECLARE @detalle NVARCHAR(255);

    IF EXISTS(SELECT * FROM inserted) AND EXISTS(SELECT * FROM deleted)
    BEGIN
        SET @accion = N'UPDATE_STOCK_BRUTO';
        SELECT @detalle = N'Materia prima "' + nombre + N'" modificada. Stock: ' + CAST(d.stock AS NVARCHAR(15)) + N' -> ' + CAST(i.stock AS NVARCHAR(15))
        FROM inserted i
        INNER JOIN deleted d ON i.id = d.id;
    END
    ELSE IF EXISTS(SELECT * FROM inserted)
    BEGIN
        SET @accion = N'INSERT_STOCK_BRUTO';
        SELECT @detalle = N'Registrada nueva materia prima "' + nombre + N'" con Stock inicial: ' + CAST(stock AS NVARCHAR(15))
        FROM inserted;
    END
    ELSE IF EXISTS(SELECT * FROM deleted)
    BEGIN
        SET @accion = N'DELETE_STOCK_BRUTO';
        SELECT @detalle = N'Materia prima "' + nombre + N'" eliminada físicamente.'
        FROM deleted;
    END;

    INSERT INTO dbo.bitacora (id_usuario, accion, accion_detalle, ip, fecha, hora_inicio, hora_cierre, created_at, updated_at)
    VALUES (NULL, @accion, @detalle, N'127.0.0.1', CAST(GETDATE() AS DATE), CAST(GETDATE() AS TIME), CAST(GETDATE() AS TIME), GETDATE(), GETDATE());
END;
GO

-- Procedimiento 1: Abrir Caja
CREATE PROCEDURE dbo.sp_AbrirCaja
    @id_usuario BIGINT,
    @monto_apertura DECIMAL(10,2),
    @monto_apertura_qr DECIMAL(10,2)
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Validar si el usuario ya tiene una caja abierta
    IF EXISTS (SELECT 1 FROM dbo.cajas WHERE id_usuario = @id_usuario AND estado = 'Abierta')
    BEGIN
        RAISERROR(N'El usuario ya tiene una sesión de caja abierta activa.', 16, 1);
        RETURN;
    END;

    INSERT INTO dbo.cajas (id_usuario, monto_apertura, monto_apertura_qr, fecha_apertura, estado, created_at, updated_at)

    DECLARE @id_caja BIGINT = SCOPE_IDENTITY();

    -- Auditoría
    INSERT INTO dbo.bitacora (id_usuario, accion, accion_detalle, ip, fecha, hora_inicio, hora_cierre, created_at, updated_at)

    SELECT * FROM dbo.cajas WHERE id = @id_caja;
END;
GO

-- Procedimiento 2: Cierre de Caja con Reporte Cuadrante Financiero
CREATE PROCEDURE dbo.sp_CerrarCaja
    @id_caja BIGINT,
    @monto_real DECIMAL(10,2),
    @monto_real_qr DECIMAL(10,2)
AS
BEGIN
    SET NOCOUNT ON;

    -- Validar que la caja esté abierta
    IF NOT EXISTS (SELECT 1 FROM dbo.cajas WHERE id = @id_caja AND estado = 'Abierta')
    BEGIN
        RAISERROR(N'La caja indicada no existe o ya se encuentra cerrada.', 16, 1);
        RETURN;
    END;

    -- Cálculos de saldos esperados y diferencias utilizando las funciones creadas
    DECLARE @ventas_efectivo DECIMAL(10,2) = dbo.fn_ObtenerVentasEfectivo(@id_caja);
    DECLARE @ventas_qr DECIMAL(10,2) = dbo.fn_ObtenerVentasQR(@id_caja);
    
    DECLARE @dif_efectivo DECIMAL(10,2) = dbo.fn_CalcularDiferenciaEfectivo(@id_caja, @monto_real);
    DECLARE @dif_qr DECIMAL(10,2) = dbo.fn_CalcularDiferenciaQR(@id_caja, @monto_real_qr);

    -- Actualizar el estado de la caja
    UPDATE dbo.cajas
    SET monto_cierre = @monto_real,
        monto_cierre_qr = @monto_real_qr,
        fecha_cierre = GETDATE(),
        estado = 'Cerrada',
        updated_at = GETDATE()
    WHERE id = @id_caja;

    DECLARE @id_usuario BIGINT;
    SELECT @id_usuario = id_usuario FROM dbo.cajas WHERE id = @id_caja;

    -- Auditoría
    DECLARE @log_detalle NVARCHAR(255) = N'Cerrada Caja ID: ' + CAST(@id_caja AS NVARCHAR(10)) + N'. Dif.Efectivo: ' + CAST(@dif_efectivo AS NVARCHAR(15)) + N' | Dif.QR: ' + CAST(@dif_qr AS NVARCHAR(15));
    INSERT INTO dbo.bitacora (id_usuario, accion, accion_detalle, ip, fecha, hora_inicio, hora_cierre, created_at, updated_at)
    VALUES (@id_usuario, N'CIERRE_CAJA', @log_detalle, N'127.0.0.1', CAST(GETDATE() AS DATE), CAST(GETDATE() AS TIME), CAST(GETDATE() AS TIME), GETDATE(), GETDATE());

    -- Retornar Reporte Comparativo Completo
    SELECT 
        c.id AS id_caja,
        c.fecha_apertura,
        c.fecha_cierre,
        c.monto_apertura AS AperturaEfectivo,
        @ventas_efectivo AS VentasEfectivoEsperadas,
        (c.monto_apertura + @ventas_efectivo) AS SaldoEfectivoEsperado,
        @monto_real AS SaldoEfectivoRealEntregado,
        @dif_efectivo AS DiferenciaEfectivo,
        c.monto_apertura_qr AS AperturaQR,
        @ventas_qr AS VentasQREsperadas,
        (c.monto_apertura_qr + @ventas_qr) AS SaldoQREsperado,
        @monto_real_qr AS SaldoQRRealEntregado,
        @dif_qr AS DiferenciaQR,
        CASE 
            WHEN @dif_efectivo = 0 AND @dif_qr = 0 THEN N'Caja Cuadrada Perfecta'
            ELSE N'Caja con Novedades'
        END AS EstadoArqueo;
END;
GO

-- Procedimiento 3: Registrar Venta con Detalles en Formato JSON
-- Esto encapsula la validación de caja, correlativos, cálculos, inserción de detalles
-- y manejo transaccional en un solo bloque. El stock se descuenta por Trigger.
CREATE PROCEDURE dbo.sp_RegistrarVenta
    @id_usuario BIGINT,
    @metodo_pago NVARCHAR(20),
    @codigo_qr NVARCHAR(100) = NULL,
    @tipo_entrega NVARCHAR(20),
    @detalles_json NVARCHAR(MAX) -- Formato: [{"id_producto": 1, "cantidad": 2, "precio_unitario": 25.00}]
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Validar si el usuario tiene una caja abierta
    DECLARE @id_caja BIGINT;
    SELECT TOP 1 @id_caja = id 
    FROM dbo.cajas 
    WHERE id_usuario = @id_usuario AND estado = 'Abierta' 
    ORDER BY fecha_apertura DESC;

    IF @id_caja IS NULL
    BEGIN
        RAISERROR(N'No se puede registrar la venta. El usuario no cuenta con una caja abierta actualmente.', 16, 1);
        RETURN;
    END;

    -- Inicio del bloque transaccional seguro
    BEGIN TRANSACTION;

    BEGIN TRY
        -- 2. Calcular el monto total sumando los productos del JSON
        DECLARE @monto_total DECIMAL(10,2);
        SELECT @monto_total = SUM(cantidad * precio_unitario)
        FROM OPENJSON(@detalles_json)
        WITH (
            cantidad INT '$.cantidad',
            precio_unitario DECIMAL(10,2) '$.precio_unitario'
        );

        -- 3. Calcular número de pedido correlativo del día
        DECLARE @nro_pedido INT;
        SELECT @nro_pedido = ISNULL(COUNT(*), 0) + 1
        FROM dbo.ventas
        WHERE CAST(created_at AS DATE) = CAST(GETDATE() AS DATE);

        -- 4. Insertar la Venta principal
        -- Al insertarse la venta, el trigger trg_CrearComandaAlVender creará la comanda de cocina automáticamente.
        DECLARE @id_venta BIGINT;
        INSERT INTO dbo.ventas (id_caja, id_usuario, monto_total, metodo_pago, codigo_qr, tipo_entrega, estado, nro_pedido, VentaEstado, created_at, updated_at)
        VALUES (@id_caja, @id_usuario, @monto_total, @metodo_pago, @codigo_qr, @tipo_entrega, 'Pagado', @nro_pedido, N'Completado', GETDATE(), GETDATE());

        SET @id_venta = SCOPE_IDENTITY();

        -- 5. Insertar los detalles de venta
        -- Esta inserción disparará el trigger trg_DescontarStockAlVender.
        -- Si hay stock insuficiente, el trigger lanzará un error, abortando la transacción entera (Rollback).
        INSERT INTO dbo.venta_detalles (id_venta, id_producto, cantidad, precio_unitario, subtotal, created_at, updated_at)
        SELECT @id_venta, id_producto, cantidad, precio_unitario, (cantidad * precio_unitario), GETDATE(), GETDATE()
        FROM OPENJSON(@detalles_json)
        WITH (
            id_producto BIGINT '$.id_producto',
            cantidad INT '$.cantidad',
            precio_unitario DECIMAL(10,2) '$.precio_unitario'
        );

        -- Registrar en bitácora
        DECLARE @accion_detalles NVARCHAR(255) = N'Venta registrada. Venta ID: ' + CAST(@id_venta AS NVARCHAR(10)) + N', Total: ' + CAST(@monto_total AS NVARCHAR(15)) + N' Bs. Pedido: #' + CAST(@nro_pedido AS NVARCHAR(10));
        INSERT INTO dbo.bitacora (id_usuario, accion, accion_detalle, ip, fecha, hora_inicio, hora_cierre, created_at, updated_at)
        VALUES (@id_usuario, N'REGISTRO_VENTA', @accion_detalles, N'127.0.0.1', CAST(GETDATE() AS DATE), CAST(GETDATE() AS TIME), CAST(GETDATE() AS TIME), GETDATE(), GETDATE());

        -- Confirmar todas las operaciones si no hubo errores
        COMMIT TRANSACTION;

        -- Retornar la venta creada y el estado de su comanda de cocina generada automáticamente
        SELECT 
            v.id AS id_venta,
            v.nro_pedido,
            v.monto_total,
            v.metodo_pago,
            v.tipo_entrega,
            c.id AS id_comanda,
            c.estado AS estado_comanda,
            c.area AS area_comanda
        FROM dbo.ventas v
        INNER JOIN dbo.comandas c ON v.id = c.id_venta
        WHERE v.id = @id_venta;

    END TRY
    BEGIN CATCH
        -- Deshacer cualquier cambio en caso de error (ej. stock insuficiente arrojado en el trigger)
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
            
        DECLARE @ErrMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrState INT = ERROR_STATE();
        
        RAISERROR(@ErrMessage, @ErrSeverity, @ErrState);
    END CATCH;
END;
GO

-- Procedimiento 4: Transformación de Inventario (Bruto -> Procesado) (CU38)
-- Permite tomar materia prima y convertirla a insumos procesados según ficha de rendimiento
CREATE PROCEDURE dbo.sp_TransformarMateriaPrima
    @id_usuario BIGINT,
    @id_bruto BIGINT,
    @id_procesado BIGINT,
    @cantidad_bruto_a_descontar DECIMAL(10,2)
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Validar existencia de la ficha de transformación
    DECLARE @cant_bruto_ficha DECIMAL(10,2);
    DECLARE @cant_procesado_ficha DECIMAL(10,2);

    SELECT 
        @cant_bruto_ficha = cantidad_bruto,
        @cant_procesado_ficha = cantidad_procesado
    FROM dbo.fichas_transformacion
    WHERE id_bruto = @id_bruto AND id_procesado = @id_procesado;

    IF @cant_bruto_ficha IS NULL
    BEGIN
        RAISERROR(N'No existe una ficha de transformación definida para estos elementos de inventario.', 16, 1);
        RETURN;
    END;

    -- 2. Validar que la materia prima tenga stock suficiente
    DECLARE @stock_bruto_actual DECIMAL(10,2);
    SELECT @stock_bruto_actual = stock FROM dbo.inventario_bruto WHERE id = @id_bruto;

    IF @stock_bruto_actual < @cantidad_bruto_a_descontar
    BEGIN
        RAISERROR(N'Stock insuficiente en inventario bruto (materia prima) para realizar la transformación.', 16, 1);
        RETURN;
    END;

    BEGIN TRANSACTION;

    BEGIN TRY
        -- Calcular la cantidad procesada generada
        DECLARE @rendimiento DECIMAL(10,4) = @cant_procesado_ficha / @cant_bruto_ficha;
        DECLARE @procesado_generado DECIMAL(10,2) = @cantidad_bruto_a_descontar * @rendimiento;

        -- Decrementar materia prima
        UPDATE dbo.inventario_bruto
        SET stock = stock - @cantidad_bruto_a_descontar,
            updated_at = GETDATE()
        WHERE id = @id_bruto;

        -- Incrementar insumos de cocina procesados
        -- Nota: La actualización del stock en estas tablas disparará automáticamente
        -- los triggers de auditoría trg_AuditarInventarioBruto y trg_AuditarInventarioProcesado.
        UPDATE dbo.inventario_procesado
        SET stock = stock + @procesado_generado,
            updated_at = GETDATE()
        WHERE id = @id_procesado;

        -- Registrar en bitácora de auditoría
        DECLARE @log_info NVARCHAR(255) = N'Transformación: -' + CAST(@cantidad_bruto_a_descontar AS NVARCHAR(15)) + N' Bruto -> +' + CAST(@procesado_generado AS NVARCHAR(15)) + N' Procesado.';
        INSERT INTO dbo.bitacora (id_usuario, accion, accion_detalle, ip, fecha, hora_inicio, hora_cierre, created_at, updated_at)
        VALUES (@id_usuario, N'PRODUCCION_TRANSFORMACION', @log_info, N'127.0.0.1', CAST(GETDATE() AS DATE), CAST(GETDATE() AS TIME), CAST(GETDATE() AS TIME), GETDATE(), GETDATE());

        COMMIT TRANSACTION;

        -- Consultar nuevos stocks resultantes
        SELECT 
            b.nombre AS MateriaPrima,
            b.stock AS StockBrutoRestante,
            p.nombre AS InsumoCocina,
            p.stock AS StockProcesadoNuevo,
            @procesado_generado AS RendimientoGenerado
        FROM dbo.inventario_bruto b
        CROSS JOIN dbo.inventario_procesado p
        WHERE b.id = @id_bruto AND p.id = @id_procesado;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
            
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        RAISERROR(@ErrorMessage, 16, 1);
    END CATCH;
END;
GO
